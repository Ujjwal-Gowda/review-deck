Building secure internal admin tools: user impersonation for support, audit logging, manual subscription overrides, and data deletion.

export async function impersonateUser(adminUserId: string, targetUserId: string) {
  await db.auditLogs.create({
    data: { adminId: adminUserId, action: "impersonate_user", targetId: targetUserId }
  });
  // Issue temporary session token with impersonation claim
  return createSessionToken({ userId: targetUserId, isImpersonated: true, originalAdminId: adminUserId });
}

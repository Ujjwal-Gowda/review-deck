2. Prerequisites & Mental Model

Node.js is an open-source, cross-platform JavaScript runtime environment built on Google Chrome's V8 engine and the Libuv asynchronous I/O library. It allows JavaScript to execute outside the browser as a systems language:

  • No DOM or Window: Global variables like window, document, and navigator do not exist in Node.js. The universal global object is globalThis (aliased as global in Node).
  • Direct OS Access: Node.js programs have full access to local files, network sockets, environment variables (process.env), command-line arguments (process.argv), and child processes.
  • Non-Blocking I/O: All slow operations (reading a file, querying a database, listening on a port) are non-blocking by default, keeping the V8 single thread responsive to new requests.

3. Module Systems: ECMAScript Modules (ESM) vs CommonJS (CJS)

Modern Node.js supports two module systems. Understanding the exact differences is critical:

Feature ECMAScript Modules (ESM) CommonJS (CJS)
Syntax import { x } from "./mod.js";
export const x = 1;		 const { x } = require("./mod");
module.exports = { x };
Loading Model Asynchronous & Static: Imports are parsed and resolved before any code executes. Tree-shaking is possible. Synchronous & Dynamic: require() executes synchronously at runtime wherever called.
Top-Level await Supported natively in ESM files. Unsupported (throws SyntaxError).
Special Variables import.meta.url (use node:url to get file path). __dirname, __filename.
File Extension / Config .mjs or "type": "module" in package.json. .cjs or default in Node without "type": "module". 
// Modern ESM standard (package.json has "type": "module")
import { readFile } from "node:fs/promises";
import { fileURLToPath } from "node:url";
import { dirname, join } from "node:path";

// Equivalent of __dirname in ESM:
const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);

const configPath = join(__dirname, "config.json");
const rawData = await readFile(configPath, "utf-8"); // Top-level await!
const config = JSON.parse(rawData);

4. Working with Large Files: Streams & Backpressure

Reading a 2GB file into memory with fs.readFile() will crash Node with an OutOfMemory error. Instead, use Streams to process data chunk-by-chunk with constant memory consumption:

  • ReadableStream: Produces chunks of data (e.g. reading from disk or incoming HTTP request).
  • WritableStream: Consumes chunks of data (e.g. writing to disk or outgoing HTTP response).
  • TransformStream: Modifies chunks in flight (e.g. compression, encryption, CSV parsing).
  • Backpressure: If the reader produces data faster than the writer can consume it, backpressure signals the reader to pause until the writer flushes its internal buffer.
import { createReadStream, createWriteStream } from "node:fs";
import { createGzip } from "node:zlib";
import { pipeline } from "node:stream/promises";

async function compressLogFile(sourcePath, destPath) {
  try {
    // stream.pipeline handles error cleanup and backpressure automatically!
    await pipeline(
      createReadStream(sourcePath), // Readable stream
      createGzip(),                 // Transform stream (compresses to .gz)
      createWriteStream(destPath)   // Writable stream
    );
    console.log("Compression completed with minimal RAM usage!");
  } catch (err) {
    console.error("Pipeline failed:", err);
  }
}

5. Zero-Dependency HTTP Server in Pure Node.js

You do not need third-party dependencies to build a production REST API. Node's built-in node:http module provides complete HTTP/1.1 capabilities:

import { createServer } from "node:http";

const PORT = 3000;
const users = new Map([
  ["1", { id: "1", name: "Alice", role: "admin" }],
  ["2", { id: "2", name: "Bob", role: "user" }]
]);

// Helper to parse incoming JSON request body
function parseJsonBody(req) {
  return new Promise((resolve, reject) => {
    let body = "";
    req.on("data", chunk => {
      body += chunk;
      // Guard against memory flood (limit body to 1MB)
      if (body.length > 1e6) {
        req.destroy();
        reject(new Error("Request payload too large"));
      }
    });
    req.on("end", () => {
      try {
        resolve(body ? JSON.parse(body) : {});
      } catch (err) {
        reject(new Error("Malformed JSON"));
      }
    });
    req.on("error", reject);
  });
}

const server = createServer(async (req, res) => {
  const url = new URL(req.url, `http://${req.headers.host}`);
  const method = req.method;

  // Set standard JSON headers
  res.setHeader("Content-Type", "application/json; charset=utf-8");

  try {
    // Route 1: GET /api/users
    if (method === "GET" && url.pathname === "/api/users") {
      res.writeHead(200);
      res.end(JSON.stringify(Array.from(users.values())));
      return;
    }

    // Route 2: GET /api/users/:id
    if (method === "GET" && url.pathname.startsWith("/api/users/")) {
      const id = url.pathname.split("/")[3];
      const user = users.get(id);
      if (!user) {
        res.writeHead(404);
        res.end(JSON.stringify({ error: "User not found" }));
        return;
      }
      res.writeHead(200);
      res.end(JSON.stringify(user));
      return;
    }

    // Route 3: POST /api/users
    if (method === "POST" && url.pathname === "/api/users") {
      const payload = await parseJsonBody(req);
      if (!payload.name) {
        res.writeHead(400);
        res.end(JSON.stringify({ error: "Property 'name' is required" }));
        return;
      }
      const newId = String(users.size + 1);
      const newUser = { id: newId, name: payload.name, role: payload.role || "user" };
      users.set(newId, newUser);
      res.writeHead(201);
      res.end(JSON.stringify(newUser));
      return;
    }

    // Default 404
    res.writeHead(404);
    res.end(JSON.stringify({ error: "Route not found" }));
  } catch (err) {
    res.writeHead(500);
    res.end(JSON.stringify({ error: err.message }));
  }
});

server.listen(PORT, () => {
  console.log(`Server listening at http://localhost:${PORT}`);
});

6. Production Reliability: Graceful Shutdown & Signal Handling

When deploying to Docker, Kubernetes, or systemd, the OS sends signals when stopping a container. Your server must intercept these signals, stop accepting new connections, finish inflight requests, and close database handles cleanly:

function setupGracefulShutdown(server) {
  const shutdown = (signal) => {
    console.log(`\nReceived ${signal}. Draining connections gracefully...`);
    
    // Stop accepting new incoming TCP connections
    server.close(() => {
      console.log("All pending requests closed. Process exiting cleanly.");
      process.exit(0);
    });

    // Forced timeout: If inflight requests do not drain in 10s, force kill
    setTimeout(() => {
      console.error("Graceful shutdown timeout exceeded. Forcefully killing process.");
      process.exit(1);
    }, 10000).unref(); // unref prevents this timer from keeping event loop alive
  };

  process.on("SIGTERM", () => shutdown("SIGTERM"));
  process.on("SIGINT", () => shutdown("SIGINT"));
}

Drill 1: ESM creates live bindings. If Module A accesses an export from Module B before Module B finishes evaluation, it encounters the Temporal Dead Zone (TDZ) and throws a ReferenceError.

Drill 2: An unhandled Promise rejection triggers the unhandledRejection event. In modern Node.js, an unhandled rejection also prints a fatal warning and will terminate the process with non-zero exit code unless handled.

PYTHON: Everything is a Heap Object x = 42 ──► [PyObject header | refcount=1 | type=&PyLong_Type | ob_digit=42] (Variables are references/names bound to heap objects) JAVASCRIPT / TYPESCRIPT: V8 Heap & Primitives let x = 42 ──► In V8: Small integers (Smi) encoded inline in pointer bits; let o = {} ──► Objects allocated on V8 Heap with Hidden Classes (Shape) GO: Direct Memory & Value Semantics var x int = 42 ──► Exact 8 bytes placed directly on the Goroutine STACK. var s MyStruct ──► Contiguous fields on stack; escapes to heap only if pointer outlives frame

3. Variables, Types, and Immutability in Detail

Python

# Dynamically typed: types are attached to runtime values, not the variable name
x = 42         # x points to an int object
x = "hello"    # perfectly legal at runtime, but discouraged in typed code

# Optional static type hints checked by pyright or mypy
count: int = 10
items: list[str] = ["apple", "banana"]

JavaScript vs TypeScript

// JavaScript: dynamic and weakly typed
const pi = 3.14159;  // cannot be reassigned
let counter = 0;     // reassignable
// counter = "done"; // Valid JS, but TypeScript compiler rejects this:

// TypeScript: static, erased type system
let typedCounter: number = 0;
// typedCounter = "done"; // Error: Type 'string' is not assignable to type 'number'

// TypeScript Interfaces and Discriminated Unions
interface User {
  id: number;
  name: string;
  role: "admin" | "member"; // Union of string literals
}

Go

// Statically and strongly typed. Zero-initialized by default!
var count int            // defaults to 0 (no uninitialized garbage memory)
var name string          // defaults to ""
var active bool          // defaults to false

// Short variable declaration (type inferred inside functions):
total := 100.5           // float64 inferred
const MaxRetries = 3     // untyped constant until used

4. Functions, Scope, and Closures

All four languages treat functions as first-class citizens (they can be stored in variables, passed into other functions as arguments, and returned from functions):

# Python closure: remembers outer variables
def make_counter(start: int = 0):
    val = start
    def step() -> int:
        nonlocal val     # required in Python to modify outer scope
        val += 1
        return val
    return step

c = make_counter(10)
print(c()) # 11
print(c()) # 12
// TypeScript closure & arrow function
function makeCounter(start: number = 0): () => number {
  let val = start;
  return () => ++val; // outer variable 'val' captured by reference
}

const count = makeCounter(10);
console.log(count()); // 11
console.log(count()); // 12
// Go closure: outer variable escapes to heap automatically
func makeCounter(start int) func() int {
    val := start
    return func() int {
        val++
        return val
    }
}

// In main():
count := makeCounter(10)
fmt.Println(count()) // 11
fmt.Println(count()) // 12

5. Error Handling: Exceptions vs Explicit Return Values

Python & JavaScript/TypeScript: Exception Trapping

try:
    with open("config.json") as f:
        data = json.load(f)
except FileNotFoundError as err:
    print(f"File missing: {err}")
except json.JSONDecodeError as err:
    print(f"Invalid JSON: {err}")
finally:
    print("Always executed for cleanup")

Go: Explicit Error Values (No Exceptions)

// In Go, functions return (Result, error). Errors are ordinary values!
func LoadConfig(path string) (*Config, error) {
    bytes, err := os.ReadFile(path)
    if err != nil {
        return nil, fmt.Errorf("reading config failed: %w", err)
    }
    var cfg Config
    if err := json.Unmarshal(bytes, &cfg); err != nil {
        return nil, fmt.Errorf("parsing config failed: %w", err)
    }
    return &cfg, nil
}

// Caller checks explicitly:
cfg, err := LoadConfig("config.json")
if err != nil {
    log.Fatalf("Fatal: %v", err)
}

6. Concurrency: Event Loops vs Goroutines

JavaScript / TypeScript (Single Thread + Event Loop) [Call Stack] ──► [WebAPIs / Libuv] ──► [Microtask Queue] (Promises) ──► [Macrotask Queue] (Timers) Go (M:N Preemptive Scheduler) Goroutine 1 ──┐ Goroutine 2 ──┼──► [Logical Processor P] ──► [OS Thread M] ──► [CPU Core] Goroutine 3 ──┘ (Runs thousands of concurrent goroutines with 2KB stack)
  • Python: Has a Global Interpreter Lock (GIL) in CPython. For I/O concurrency, use asyncio (cooperative single-threaded event loop). For multi-core CPU parallelism, use multiprocessing.
  • JavaScript / TypeScript: Runs on a single OS thread with an asynchronous Event Loop. Long CPU computations block the UI or server; I/O is completely non-blocking via Promises and async/await.
  • Go: Uses lightweight Goroutines (costing only ~2KB of initial stack memory vs 2MB for an OS thread). Multiplexes millions of goroutines across all CPU cores with channels for safe synchronization (CSP).

Rust's syntax and types: variables, mutability, shadowing, scalar and compound types, and expressions.

A first program

use std::io::{self, BufRead};

const TAX_PERCENT: u64 = 5;

fn with_tax(paise: u64) -> u64 {
    paise * (100 + TAX_PERCENT) / 100          // no semicolon = this expression is the return value
}

fn main() {
    let stdin = io::stdin();
    let mut total: u64 = 0;
    for line in stdin.lock().lines() {
        let line = line.expect("read error");  // shadowing: a new `line` of a different type
        match line.trim().parse::<u64>() {
            Ok(p) => total += p,
            Err(e) => eprintln!("skipping {line:?}: {e}"),
        }
    }
    println!("total with tax: ₹{:.2}", with_tax(total) as f64 / 100.0);
}

Variables

let qty = 2;            // immutable by default
// qty = 3;             // error: cannot assign twice to immutable variable
let mut total = 0;      // mutable
total += qty;
let qty = qty * 10;     // SHADOWING: a new binding (can even change type), handy for transformations
let price: u32;         // declare now, assign once later (the compiler checks it's set before use)

Defining your own types in Python, and hooking into the language through the data model ("dunder" methods).

A class from scratch

class Cart:
    tax_rate = 0.05                          # class attribute (shared by all instances)

    def __init__(self, customer: str):
        self.customer = customer             # instance attributes
        self._items: dict[str, int] = {}     # _ = internal by convention

    def add(self, sku: str, qty: int = 1) -> None:
        if qty <= 0: raise ValueError("qty must be positive")
        self._items[sku] = self._items.get(sku, 0) + qty

    @property
    def count(self) -> int:                  # accessed like an attribute: cart.count
        return sum(self._items.values())

    def __len__(self): return len(self._items)          # len(cart)
    def __contains__(self, sku): return sku in self._items   # "milk" in cart
    def __iter__(self): return iter(self._items.items())    # for sku, qty in cart
    def __repr__(self): return f"Cart({self.customer!r}, {self._items})"

Dataclasses: less boilerplate

from dataclasses import dataclass, field
@dataclass(frozen=True, slots=True)          # immutable, hashable, memory-efficient
class Money:
    paise: int
    def __add__(self, other: "Money") -> "Money": return Money(self.paise + other.paise)
    def __str__(self): return f"₹{self.paise / 100:,.2f}"

@dataclass(order=True)                       # generates __init__, __repr__, __eq__, and < > comparisons
class Delivery:
    deadline: int
    order_id: int = field(compare=False)
    tags: list[str] = field(default_factory=list, compare=False)

Inheritance and protocols

from abc import ABC, abstractmethod
class Notifier(ABC):
    @abstractmethod
    def send(self, to: str, msg: str) -> None: ...
class Sms(Notifier):
    def send(self, to, msg): print(f"SMS to {to}: {msg}")

from typing import Protocol                  # structural: no inheritance needed
class SupportsSend(Protocol):
    def send(self, to: str, msg: str) -> None: ...
def notify_all(n: SupportsSend, phones): [n.send(p, "Sale!") for p in phones]

Use super().__init__(…) to call the parent's constructor, and prefer composition over deep hierarchies (OOP).

In practice

Plain data: @dataclass (frozen when possible). Validation of external data: Pydantic models. Behaviour-only helpers: plain functions, not classes. Interfaces: Protocol.

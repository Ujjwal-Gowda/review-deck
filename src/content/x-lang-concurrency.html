async function fetchAll(urls: string[], limit = 5) {
  const results: number[] = []; let i = 0;
  const worker = async () => { while (i < urls.length) { const k = i++; results[k] = (await fetch(urls[k])).status; } };
  await Promise.all(Array.from({ length: limit }, worker));
  return results;
}
func fetchAll(ctx context.Context, urls []string) []int {
    out := make([]int, len(urls))
    g, ctx := errgroup.WithContext(ctx); g.SetLimit(5)
    for i, u := range urls {
        g.Go(func() error { req, _ := http.NewRequestWithContext(ctx, "GET", u, nil)
            r, err := http.DefaultClient.Do(req); if err != nil { return err }
            r.Body.Close(); out[i] = r.StatusCode; return nil })
    }
    g.Wait(); return out
}
async fn fetch_all(urls: Vec<String>) -> Vec<u16> {   // tokio + reqwest + futures
    let client = reqwest::Client::new();
    futures::stream::iter(urls).map(|u| { let c = client.clone(); async move {
        c.get(&u).send().await.map(|r| r.status().as_u16()).unwrap_or(0) } })
      .buffered(5).collect().await                    // at most 5 in flight, results in order
}

0.1 basic Sum the squares of 1..50,000,000 using all CPU cores in Go (goroutines), Rust (rayon par_iter) and Python (ProcessPoolExecutor). Compare wall time with the single-threaded version of each.

Go and Rust scale close to your 12 cores. Python with processes scales too, but pays for starting processes and shipping data between them (chunk the ranges). Python threads would show no speedup because of the GIL.

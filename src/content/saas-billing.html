Connecting subscriptions to product features: gating UI and API endpoints by active plan tiers, feature flags, and billing portals.

export async function requirePlanTier(req: Request, res: Response, next: NextFunction) {
  const team = req.team;
  if (!team.plan || team.plan === "free") {
    return res.status(403).json({
      error: "Upgrade required",
      upgrade_url: "/settings/billing"
    });
  }
  next();
}

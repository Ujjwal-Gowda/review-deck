Vector Generation and Math

import numpy as np
from openai import OpenAI
client = OpenAI()

def get_embedding(text: str, model="text-embedding-3-small") -> list[float]:
    cleaned = text.replace("
", " ")
    res = client.embeddings.create(input=[cleaned], model=model)
    return res.data[0].embedding

def cosine_similarity(a: list[float], b: list[float]) -> float:
    v1, v2 = np.array(a), np.array(b)
    return float(np.dot(v1, v2) / (np.linalg.norm(v1) * np.linalg.norm(v2)))

def batch_embed(texts: list[str]) -> list[list[float]]:
    res = client.embeddings.create(input=texts, model="text-embedding-3-small")
    return [item.embedding for item in res.data]

Lifecycle Management in PostgreSQL (pgvector)

CREATE EXTENSION IF NOT EXISTS vector;

CREATE TABLE articles (
    id SERIAL PRIMARY KEY,
    title TEXT NOT NULL,
    content TEXT NOT NULL,
    content_hash CHAR(64) NOT NULL,
    embedding vector(1536)
);

CREATE INDEX ON articles USING hnsw (embedding vector_cosine_ops);
Warning · Staleness & Dimension Mismatches

If you edit document text without updating its embedding, search results degrade silently. Always attach a content hash or use an async DB trigger to mark embeddings as dirty.

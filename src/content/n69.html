Narrowing and type guards

function fmt(x: string | number | Date) {
  if (typeof x === "string") return x;            // typeof guard
  if (x instanceof Date) return x.toISOString();  // instanceof guard
  return x.toFixed(2);                            // TS knows only number is left
}
function isOrder(v: unknown): v is Order {         // custom type predicate
  return typeof v === "object" && v !== null && "id" in v && "items" in v;
}
function assertNever(x: never): never { throw new Error(`unhandled: ${JSON.stringify(x)}`); }
// in a switch's default: assertNever(s) → compile error if you forgot a union member

Type-level programming

type Keys = keyof Order;                                  // "id" | "items" | "status"
type Status = Order["status"];                            // indexed access
const ROLES = ["owner", "manager", "cashier"] as const;
type Role = (typeof ROLES)[number];                       // "owner" | "manager" | "cashier"
type Optional<T, K extends keyof T> = Omit<T, K> & Partial<Pick<T, K>>;   // make some keys optional
type Readonlyish<T> = { readonly [K in keyof T]: T[K] };  // mapped type
type Unwrap<T> = T extends Promise<infer U> ? U : T;       // conditional type + infer
type Route = `/api/${"orders" | "stock"}/${number}`;       // template literal type
type Handlers = { [K in Status as `on${Capitalize<K>}`]: () => void };   // key remapping
function config<const T extends Record<string, unknown>>(c: T) { return c; }
const cfg = { port: 8080 } satisfies Record<string, number>;   // checks the type without widening it

Runtime validation: types disappear at runtime

import { z } from "zod";
const OrderSchema = z.object({ id: z.number().int(), items: z.array(z.object({ sku: z.string(), qty: z.number().positive() })) });
type Order = z.infer<typeof OrderSchema>;                  // ONE source of truth for type AND validator
const order = OrderSchema.parse(await res.json());         // throws on bad data at the boundary

1. The Zero-Decimal Currency Trap

Most currencies (USD, EUR, GBP, INR) have 100 subunits per unit (cents, pence, paise). However, several currencies are zero-decimal—they have no smaller unit!

CurrencyDecimal PlacesExample: 10 units in smallest denomination
USD, EUR, INR2 decimals1000 ($10.00 / €10.00 / ₹10.00)
JPY (Japanese Yen)0 decimals10 (¥10)
KRW (Korean Won)0 decimals10 (₩10)
BHD, KWD (Dinars)3 decimals10000 (10.000 BD)
const ZERO_DECIMAL_CURRENCIES = new Set(["JPY", "KRW", "BIF", "CLP", "DJF", "GNF", "PYG", "RWF", "UGX", "VND", "VUV"]);

export function toGatewayAmount(amount: number, currency: string): number {
  const code = currency.toUpperCase();
  if (ZERO_DECIMAL_CURRENCIES.has(code)) {
    return Math.round(amount);
  }
  return Math.round(amount * 100);
}

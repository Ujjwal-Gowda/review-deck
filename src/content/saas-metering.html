Tracking and enforcing consumption limits (API calls, storage, seats, AI tokens) using Redis sliding windows and DB counters.

export async function checkAndIncrementUsage(tenantId: string, limit: number): Promise {
  const currentMonth = new Date().toISOString().slice(0, 7); // e.g. '2026-10'
  const key = `usage:${tenantId}:${currentMonth}`;
  
  const current = await redis.incr(key);
  if (current === 1) await redis.expire(key, 60 * 60 * 24 * 35);
  
  return current <= limit;
}

Building your own MCP server, locally first, then hosted.

# notes_server.py    pip install "mcp[cli]"      run/test: mcp dev notes_server.py
from mcp.server.fastmcp import FastMCP
from pathlib import Path
mcp = FastMCP("notes")
NOTES = Path.home() / "notes"

@mcp.tool()
def search_notes(query: str, limit: int = 5) -> list[dict]:
    """Search my notes for a phrase. Returns the file name and matching line."""
    hits = []
    for f in NOTES.rglob("*.md"):
        for line in f.read_text(encoding="utf-8").splitlines():
            if query.lower() in line.lower():
                hits.append({"file": f.name, "line": line.strip()})
                if len(hits) >= limit: return hits
    return hits

@mcp.resource("notes://{name}")
def read_note(name: str) -> str:
    """Full text of one note."""
    p = (NOTES / name).resolve()
    if not p.is_relative_to(NOTES): raise ValueError("outside notes dir")   # path traversal guard (chapter 162)
    return p.read_text(encoding="utf-8")

if __name__ == "__main__":
    mcp.run()          # stdio by default; mcp.run(transport="streamable-http") to serve over HTTP
claude mcp add notes -- python3 ~/mcp/notes_server.py      # register it with Claude Code (stdio)
# TypeScript: @modelcontextprotocol/sdk   ·  Go and Rust SDKs exist too
  • Local vs hosted: a stdio server needs no hosting: the host starts it on demand. Host it (HTTP, behind TLS with OAuth) only when several users or devices need it, or when it wraps a server-side system.
  • Design tools like APIs: few, clear, well-described, validated, least-privilege, with concise outputs.
  • Test with the MCP Inspector (npx @modelcontextprotocol/inspector).

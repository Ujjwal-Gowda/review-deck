Classes & modules

class Cart {
  #items = new Map<string, number>();   // truly private
  constructor(public readonly customer: string) {}
  add(sku: string, n = 1) { this.#items.set(sku, (this.#items.get(sku) ?? 0) + n); }
  get count() { return [...this.#items.values()].reduce((a, b) => a + b, 0); }
}

Project 10: Building Production MCP Servers. Master Anthropic's Model Context Protocol (MCP) by constructing a secure, dual-transport (stdio & SSE) operational server for AI developer tools.

The Architecture of the Model Context Protocol

Before MCP, every AI agent framework (LangChain, LlamaIndex, custom scripts) invented its own proprietary tool-calling schema. If you built a tool for PostgreSQL, you had to rewrite it for Claude, rewrite it again for VS Code, and rewrite it again for custom Python agents. Model Context Protocol (MCP) is the "USB-C for AI": an open JSON-RPC 2.0 standard allowing any AI client (Antigravity, Claude Code, Cursor, Zed) to securely discover and invoke tools, read resources, and execute prompt templates.

Part 2: Building the Production TypeScript MCP Server

Create a robust DevOps & Database inspection server using the official @modelcontextprotocol/sdk:

import { Server } from "@modelcontextprotocol/sdk/server/index.js";
import { StdioServerTransport } from "@modelcontextprotocol/sdk/server/stdio.js";
import {
  CallToolRequestSchema,
  ListToolsRequestSchema,
  ListResourcesRequestSchema,
  ReadResourceRequestSchema,
  ListPromptsRequestSchema,
  GetPromptRequestSchema,
} from "@modelcontextprotocol/sdk/types.js";
import { z } from "zod";
import { Pool } from "pg";
import os from "os";

const db = new Pool({
  connectionString: process.env.READONLY_DATABASE_URL,
  max: 3,
  statement_timeout: 4000, // 4-second hard timeout to prevent runaway queries
});

const server = new Server(
  { name: "ops-nexus", version: "1.0.0" },
  { capabilities: { tools: {}, resources: {}, prompts: {} } }
);

// -------------------------------------------------------------------
// 1. List Available Tools
// -------------------------------------------------------------------
server.setRequestHandler(ListToolsRequestSchema, async () => ({
  tools: [
    {
      name: "query_readonly_sql",
      description: "Execute a read-only SELECT SQL query against the production replica. Mutating commands (INSERT, UPDATE, DELETE, DROP) are strictly blocked.",
      inputSchema: {
        type: "object",
        properties: {
          sql: { type: "string", description: "The SELECT query to execute" },
        },
        required: ["sql"],
      },
    },
    {
      name: "get_host_metrics",
      description: "Retrieve real-time host CPU load, memory utilization, and uptime.",
      inputSchema: { type: "object", properties: {} },
    },
  ],
}));

// -------------------------------------------------------------------
// 2. Execute Tools with Strict Safety Guardrails
// -------------------------------------------------------------------
server.setRequestHandler(CallToolRequestSchema, async (request) => {
  const { name, arguments: args } = request.params;

  if (name === "get_host_metrics") {
    const freeMemMb = Math.round(os.freemem() / 1024 / 1024);
    const totalMemMb = Math.round(os.totalmem() / 1024 / 1024);
    return {
      content: [
        {
          type: "text",
          text: JSON.stringify({
            hostname: os.hostname(),
            uptimeSeconds: os.uptime(),
            cpuCount: os.cpus().length,
            loadAvg1m: os.loadavg()[0],
            memoryUsedPercentage: (((totalMemMb - freeMemMb) / totalMemMb) * 100).toFixed(1) + "%",
            memoryFreeMb: freeMemMb,
          }, null, 2),
        },
      ],
    };
  }

  if (name === "query_readonly_sql") {
    const rawSql = String(args?.sql || "").trim();

    // Defense-in-depth AST & regex safety check
    const forbiddenKeywords = /\b(DROP|DELETE|UPDATE|INSERT|ALTER|TRUNCATE|GRANT|REVOKE)\b/i;
    if (forbiddenKeywords.test(rawSql) || !rawSql.toUpperCase().startsWith("SELECT")) {
      return {
        isError: true,
        content: [{ type: "text", text: "SECURITY_VIOLATION: Only SELECT statements are permitted." }],
      };
    }

    try {
      const result = await db.query(rawSql);
      return {
        content: [{ type: "text", text: JSON.stringify(result.rows, null, 2) }],
      };
    } catch (err: any) {
      return {
        isError: true,
        content: [{ type: "text", text: `Database error: ${err.message}` }],
      };
    }
  }

  throw new Error(`Unknown tool: ${name}`);
});

// -------------------------------------------------------------------
// 3. Expose Live Resources
// -------------------------------------------------------------------
server.setRequestHandler(ListResourcesRequestSchema, async () => ({
  resources: [
    {
      uri: "db://schema/public",
      name: "PostgreSQL Database Schema",
      mimeType: "application/json",
      description: "List of all public tables, columns, and foreign keys.",
    },
  ],
}));

server.setRequestHandler(ReadResourceRequestSchema, async (request) => {
  if (request.params.uri === "db://schema/public") {
    const res = await db.query(`
      SELECT table_name, column_name, data_type 
      FROM information_schema.columns 
      WHERE table_schema = 'public' 
      ORDER BY table_name, ordinal_position;
    `);
    return {
      contents: [
        {
          uri: request.params.uri,
          mimeType: "application/json",
          text: JSON.stringify(res.rows, null, 2),
        },
      ],
    };
  }
  throw new Error("Resource not found");
});

// -------------------------------------------------------------------
// Start Transport (stdio)
// -------------------------------------------------------------------
const transport = new StdioServerTransport();
await server.connect(transport);

The Stdio stdout Corruption Bug

When running an MCP server over stdio, never use console.log() for debug statements!

The MCP protocol uses process.stdout exclusively for framing JSON-RPC packets. If your server outputs console.log("Connected to DB"), the client parser fails with SyntaxError: Unexpected token 'C' and immediately terminates the process.

Always direct debug logs to standard error: console.error("[DEBUG] Connected to DB").

Add the server configuration to your project's .mcp.json or global claude.json:

{
  "mcpServers": {
    "ops-nexus": {
      "command": "node",
      "args": ["/absolute/path/to/ops-server/dist/server.js"],
      "env": {
        "READONLY_DATABASE_URL": "postgres://readonly_user:secret@db.internal:5432/production"
      }
    }
  }

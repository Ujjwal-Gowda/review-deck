Forms that are pleasant to fill in, accessible, and validated correctly on both ends.

import { useForm } from "react-hook-form";
import { zodResolver } from "@hookform/resolvers/zod";
const Schema = z.object({ phone: z.string().regex(/^[6-9]\d{9}$/, "10-digit mobile"), qty: z.coerce.number().int().min(1).max(20) });
type Values = z.infer<typeof Schema>;

function OrderForm() {
  const { register, handleSubmit, formState: { errors, isSubmitting } } = useForm<Values>({ resolver: zodResolver(Schema) });
  return (
    <form onSubmit={handleSubmit(async v => { await placeOrder(v); })} noValidate>
      <label htmlFor="phone">Mobile</label>
      <input id="phone" inputMode="numeric" autoComplete="tel" aria-invalid={!!errors.phone}
             aria-describedby="phone-err" {...register("phone")} />
      {errors.phone && <p id="phone-err" role="alert">{errors.phone.message}</p>}
      <button disabled={isSubmitting}>{isSubmitting ? "Placing…" : "Place order"}</button>
    </form>
  );
}
  • Controlled inputs (React state holds the value; re-render on every keystroke) vs uncontrolled (the DOM holds the value; read it on submit). Form libraries use uncontrolled inputs for speed.
  • Share the schema between client and server (zod works in both), but the server must validate regardless.
  • UX: validate on blur, then live once a field has been touched; keep the user's input on error; disable the submit button while pending to prevent double orders (and make the server idempotent anyway).
  • Accessibility: a real <label> on every input, errors linked with aria-describedby, focus moved to the first error on submit.

Rust's tools for modelling data precisely: structs with methods, enums that carry data, and pattern matching that checks you handled every case.

Structs and impl blocks

#[derive(Debug, Clone, PartialEq)]
pub struct Item { pub sku: String, pub qty: u32, pub price_paise: u64 }

impl Item {
    pub fn new(sku: impl Into<String>, price_paise: u64) -> Self {   // associated function (constructor)
        Self { sku: sku.into(), qty: 1, price_paise }
    }
    pub fn total(&self) -> u64 { self.qty as u64 * self.price_paise }   // &self: read
    pub fn add(&mut self, n: u32) { self.qty += n; }                   // &mut self: modify
    pub fn into_sku(self) -> String { self.sku }                      // self: consume
}
let mut it = Item::new("milk", 6000);
it.add(1);
let updated = Item { qty: 5, ..it.clone() };                          // struct update syntax

Enums with data

#[derive(Debug)]
enum OrderStatus {
    Placed,
    Paid { txn_id: String, amount: u64 },
    Shipped { tracking: String },
    Cancelled(String),                       // tuple-style variant
}
impl OrderStatus {
    fn can_cancel(&self) -> bool { matches!(self, Self::Placed | Self::Paid { .. }) }
}

fn describe(s: &OrderStatus) -> String {
    match s {
        OrderStatus::Placed => "waiting for payment".into(),
        OrderStatus::Paid { amount, .. } if *amount > 1_000_000 => "paid (large, needs review)".into(),
        OrderStatus::Paid { txn_id, .. } => format!("paid via {txn_id}"),
        OrderStatus::Shipped { tracking } => format!("shipped: {tracking}"),
        OrderStatus::Cancelled(reason) => format!("cancelled: {reason}"),
    }                              // forget a variant → compile error (exhaustiveness)
}

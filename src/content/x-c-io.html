Reading and writing files in C, with the standard library's buffered FILE* API and the lower-level POSIX calls underneath.

stdio: buffered and portable

FILE *in = fopen("stock.csv", "r");            /* modes: "r", "w" (truncate), "a" (append), add "b" for binary */
if (!in) { fprintf(stderr, "stock.csv: %s\n", strerror(errno)); return 1; }

char line[256];
int lineno = 0;
while (fgets(line, sizeof line, in)) {           /* reads up to the newline, NUL-terminates */
    lineno++;
    line[strcspn(line, "\n")] = '\0';             /* strip the newline */
    char sku[32]; int qty;
    if (sscanf(line, "%31[^,],%d", sku, &qty) != 2) {   /* width limit prevents overflow */
        fprintf(stderr, "line %d: bad format\n", lineno); continue;
    }
    /* use sku, qty */
}
if (ferror(in)) perror("read");
fclose(in);

FILE *out = fopen("report.txt", "w");
fprintf(out, "%-10s %5d\n", "milk", 12);
if (fclose(out) != 0) perror("close");            /* write errors can surface at close */

POSIX file descriptors

#include <fcntl.h>
#include <unistd.h>
int fd = open("log.txt", O_WRONLY | O_CREAT | O_APPEND, 0644);
if (fd < 0) { perror("open"); return 1; }
const char msg[] = "order placed\n";
ssize_t w = write(fd, msg, sizeof msg - 1);        /* may write FEWER bytes than asked: loop in real code */
fsync(fd);                                        /* force it to disk (durability) */
close(fd);

stdio (FILE*) buffers for you and is portable; use it by default. File descriptors are the raw OS interface: they're needed for sockets, pipes and non-blocking I/O, and they're what strace shows (I/O Models). stdin, stdout and stderr are descriptors 0, 1 and 2.
